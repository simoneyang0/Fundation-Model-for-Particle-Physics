# 1. Crea uno script Python dal tuo notebook
jupyter nbconvert --to script training.ipynb
# Questo crea training.py

# 2. Crea uno script wrapper
cat > run_training.sh << 'EOF'
#!/bin/bash
echo "Job started at $(date)"
echo "Running on node: $(hostname)"

# Attiva ambiente virtuale se necessario
source /afs/cern.ch/user/t/tuo_username/ml_env/bin/activate

# Esegui lo script Python
python training.py --epochs 50 --batch_size 32

echo "Job finished at $(date)"
EOF

# 3. Rendi eseguibile lo script
chmod +x run_training.sh

# 4. Crea il file di submit
cat > job.submit << 'EOF'
executable = run_training.sh
output = output/training.$(ClusterId).$(ProcId).out
error = error/training.$(ClusterId).$(ProcId).err
log = log/training.$(ClusterId).log
request_cpus = 4
request_memory = 12288
+JobFlavour = "workday"
should_transfer_files = YES
when_to_transfer_output = ON_EXIT

queue 1
EOF

# 5. Crea le directory necessarie
mkdir -p output error log

# 6. Sottometti il job
condor_submit job.submit
